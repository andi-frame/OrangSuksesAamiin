\chapter{Problem 3 -- Autentikasi dan Kepercayaan antar-Service}

\begin{ringkas}
Prinsipnya: \textbf{setiap domain kepercayaan punya kredensial sendiri, diterbitkan dan diperiksa oleh pemiliknya,
dan tidak ada satu rahasia pun yang dipakai di dua domain}. Kunci BMKG tidak membuka PVMBG, kunci PVMBG tidak membuka
BMKG, dan token klien tidak membuka keduanya. Untuk klien, Auth Service menerbitkan token JWT berumur 60~detik per
peran, dan Client-Facing API menegakkan hak akses \emph{di server}: Media tidak pernah menerima field Mentah.
\end{ringkas}

tetapi tidak bisa mencetak token baru atau mengadministrasi PVMBG.

\subsection{Klien downstream: Media, Tim Lapangan, BNPB Pusat}

\begin{itemize}
 \item \textbf{Tiga identitas yang benar-benar berbeda} (\code{AUTH_MEDIA_*}, \code{AUTH_LAPANGAN_*}, \code{AUTH_PUSAT_*}, masing-masing dengan id dan secret sendiri), bukan satu akun dengan bendera peran di sisi klien. Peran ditentukan Auth Service dari identitas yang lolos login, lalu ditulis ke klaim \code{role} \emph{di dalam token bertanda tangan}. Klien tidak bisa memilih perannya.
 \item \textbf{Access token:} JWT HS256 dengan klaim \code{sub}, \code{role}, \code{jti}, \code{iat}, \code{exp}. TTL \textbf{60~detik} (\code{AUTH_ACCESS_TTL_SEC}, dapat diubah).
 \item \textbf{Refresh token:} 256 bit acak, \textbf{tidak pernah disimpan polos}; yang disimpan hanya hash SHA-256 di SQLite milik Auth Service (volume yang hanya dipasang pada container itu). Sekali pakai: \cmd{POST /v1/refresh} menandai token lama \code{revoked} dalam transaksi yang sama dengan pembacaannya, lalu menerbitkan pasangan baru. TTL \code{AUTH_REFRESH_TTL_SEC} (24~jam).
 \item \textbf{Penegakan di sisi BNPB, pada setiap permintaan} (\code{services/client-api}):
  \begin{enumerate}
   \item Tanpa \cmd{Authorization: Bearer} $\rightarrow$ \textbf{401}. Token rusak, kedaluwarsa, salah tanda tangan, atau peran tak dikenal $\rightarrow$ \textbf{403}.
   \item \textbf{Media} menerima \emph{hanya} field Ringkasan lewat \emph{whitelist} (\code{policy.FilterItem}). Field baru yang kelak muncul di HazardEvent tidak otomatis bocor ke Media.
   \item Permintaan Media atas field Mentah (\code{fields=latitude}, \code{include=mentah}, dan sejenisnya) \textbf{ditolak 403}. Aturannya juga whitelist (\code{policy.CheckQuery}): setiap nilai \code{fields} harus field Ringkasan, dan \code{include} hanya boleh kosong atau \code{summary}. Variasi penulisan (huruf besar, spasi, \code{fields[]}, parameter berulang, daftar dipisah koma, nama yang tak dikenal) tidak lolos.
   \item \code{/v1/sources/status} hanya untuk Lapangan dan Pusat; Media mendapat 403.
  \end{enumerate}
 \item Kredensial, kunci penanda tangan, dan konfigurasi dibaca dari \emph{environment}. Service berhenti dengan pesan jelas bila variabelnya kosong (\code{obs.MustEnv}). Repositori hanya memuat \code{.env.example} berisi nilai contoh.
\end{itemize}

\textbf{Harga yang diterima.} \code{AUTH_JWT_SECRET} dipegang \emph{dua} service milik BNPB: Auth Service dan
Client-Facing API. Rahasia ini independen dari kredensial BMKG, PVMBG, kunci layanan, dan token admin, sehingga
kebocorannya tidak merambat ke domain lain. Konsekuensinya, bila Client-Facing API dibobol, penyerang dapat mencetak
token klien. Itulah alasan pilihan asimetris dicatat sebagai langkah berikutnya.

\section{Batasan yang disadari}\label{sec:p3-batasan}

\begin{itemize}
 \item \textbf{Access token lama tidak dicabut seketika saat refresh.} JWT bersifat \emph{stateless}, jadi token yang belum kedaluwarsa tetap berlaku sampai \code{exp} (paling lama 60~detik). Yang dicabut seketika adalah \emph{refresh token} lama (terbukti: dipakai ulang mendapat 403). Pencabutan seketika menuntut introspeksi atau daftar \code{jti} yang selalu diperbarui.
 \item \textbf{Tidak ada deteksi pemakaian ulang refresh token.} Keluarga token tidak dicabut ketika refresh yang sudah terpakai muncul lagi.
 \item \textbf{\code{/v1/token} tidak dibatasi percobaan} (tanpa lockout atau pembatasan laju per klien). Perbandingan secret memakai waktu konstan.
 \item \textbf{Pembatas laju Client-Facing API bersifat global} dan dipasang sebelum autentikasi, jadi banjir permintaan tanpa token ikut menghabiskan jatah. Per klien setelah autentikasi adalah perbaikan M2.
 \item Token kedaluwarsa dijawab \textbf{403} (bukan 401), mengikuti kontrak yang dibekukan di \code{docs/contracts.md}.
 \item Tidak ada TLS di dalam jaringan Compose; secret klien statis di \emph{environment} tanpa perkakas rotasi; SQLite Auth Service hanya cocok untuk satu instance (M2 butuh penyimpanan bersama).
\end{itemize}

\section{Jawaban butir ``Di laporan''}

\begin{jawaban}[Mekanisme yang dipakai]
Dua lapis. \textbf{(1) Antar-instansi:} kredensial BMKG (kunci buram) dan PVMBG (token bertanda tangan HMAC dengan
struktur lain; rahasia penanda tangan hanya di mock PVMBG) saling asing dan masing-masing diperiksa pemiliknya.
\textbf{(2) Klien downstream:} Auth Service menerbitkan JWT HS256 berumur 60~detik per peran ditambah refresh token buram
sekali pakai. Client-Facing API memeriksa setiap permintaan dan menegakkan klasifikasi Ringkasan/Mentah di server.
\end{jawaban}

\begin{jawaban}[Mengapa tidak membuka celah baru]
\begin{itemize}
 \item \textbf{Pemalsuan token:} algoritma dikunci ke HS256 (\code{alg=none} dan algoritma lain ditolak; diuji). \code{exp} wajib ada dan diperiksa (token bertanda tangan tanpa \code{exp} ditolak; diuji). Peran harus salah satu dari tiga nilai yang dikenal.
 \item \textbf{Eskalasi peran dari sisi klien:} peran hanya berasal dari klaim bertanda tangan. Header atau parameter buatan klien (misalnya \code{X-Role}) tidak dibaca. Media tidak punya jalur ke field Mentah karena respons dibentuk dengan whitelist, bukan dengan menghapus field yang dikenal.
 \item \textbf{Kebocoran lintas domain:} rahasia tiap domain terpisah dan tidak dipakai ulang. Aggregator tidak memegang rahasia penanda tangan PVMBG (diverifikasi \code{demo-mock-auth}).
 \item \textbf{Pencurian dari penyimpanan:} refresh token hanya tersimpan sebagai hash. Basis datanya tidak dapat dijangkau service lain. Pencurian refresh token dibatasi oleh sifat sekali pakai dan TTL.
 \item \textbf{Jendela penyalahgunaan:} token curian berguna paling lama 60~detik (access) atau sekali (refresh).
 \item \textbf{Pembatasan yang diakui} ada di bagian~\ref{sec:p3-batasan}: tidak ada pencabutan access token seketika, dan HS256 dibagi dua service BNPB.
\end{itemize}
\end{jawaban}
