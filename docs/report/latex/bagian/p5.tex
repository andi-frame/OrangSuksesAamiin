\chapter{Problem 5 -- Fan-out Event Distribution \mbox{(Pub/Sub)}}

\begin{ringkas}
Aggregator \textbf{tidak pernah memanggil \emph{consumer}}. Setiap perubahan hazard ditulis ke tabel \emph{outbox} dalam
transaksi yang sama dengan hazard-nya, lalu sebuah \emph{relay} mempublikasikannya ke \emph{topic} Kafka \code{hazard.events}.
Setiap \emph{consumer} men-\emph{subscribe} dengan \emph{consumer group} masing-masing. Pengiriman bersifat \textbf{at-least-once},
sehingga \emph{consumer} membuang duplikat berdasarkan \code{event_id}. \emph{Consumer} yang mati tidak menahan \emph{producer}, dan akan
menyusul \emph{event} yang terlewat saat hidup kembali.
\end{ringkas}

\begin{catatan}[Status bukti Problem 5]
Keempat kriteria dibuktikan oleh satu skrip, \cmd{make demo-p5}: \emph{event} lewat \emph{broker}, dua \emph{consumer} independen, \emph{consumer}
yang mati lalu menyusul \emph{dengan ID event yang tepat}, dan \emph{consumer} ketiga (\code{pemda-portal}) yang ditambahkan
saat sistem berjalan tanpa menyentuh Aggregator. \emph{Consumer} ketiga ini tidak ada di Compose, dan dinyalakan oleh skrip
demo untuk membuktikan bahwa menambah \emph{consumer} tidak membutuhkan perubahan pada \emph{producer}.
\end{catatan}

\subsection{Sisi producer: transactional outbox}

\begin{enumerate}
 \item Saat Aggregator menyimpan hazard baru atau versi baru, ia \emph{juga} menulis satu baris \code{outbox} dalam \textbf{transaksi database yang sama}. Keduanya berhasil bersama atau gagal bersama, sehingga tidak ada hazard yang tersimpan tanpa \emph{event}.
 \item \textbf{\emph{Relay}} (\code{aggregator/internal/relay}) membaca baris \emph{outbox} yang belum terkirim, mengirimnya ke Kafka berurutan dari yang paling lama, dan baru menandainya terkirim \emph{setelah} Kafka mengakuinya. Jika satu pengiriman gagal, \emph{relay} berhenti di titik itu (untuk menjaga urutan per hazard), menunggu dengan jeda yang makin panjang, lalu mencoba lagi.
 \item Akibatnya, jika Kafka mati, ingest tidak berhenti dan tidak ada \emph{event} yang hilang. \emph{Event} menunggu di \emph{outbox} dan terkirim berurutan setelah \emph{broker} pulih (\code{TestBrokerOutageDoesNotStopIngestionOrLoseEvents}).
 \item \emph{Producer} Kafka memakai \code{acks=all} dan mode idempoten (\code{franz-go}).
\end{enumerate}

\subsection{Sisi consumer}

Setiap \emph{consumer} (\code{services/consumer-hazard}) men-\emph{subscribe} \emph{topic} dengan group-nya sendiri. Untuk tiap pesan, \emph{consumer}
(1) membaca \emph{header} untuk melanjutkan \code{cid}, (2) membuang pesan yang \code{event_id}-nya sudah pernah diproses,
(3) memproses pesan, lalu (4) menandai \emph{offset} untuk di-\emph{commit}. Karena \emph{offset} ditandai \emph{setelah} pesan
diproses, pesan tidak pernah hilang jika \emph{consumer} mati di tengah jalan. Dalam kasus terburuk, pesan itu diproses lagi.

\subsection{Semantik pengiriman: at-least-once}

\begin{itemize}
 \item \emph{Producer} ke Kafka: \emph{relay} menandai baris terkirim hanya setelah mendapat pengakuan. Jika Aggregator mati \emph{tepat di antara} pengakuan dan penandaan, baris yang sama dikirim ulang. Satu \emph{event} bisa tiba dua kali, tetapi tidak pernah nol kali.
 \item Kafka ke \emph{consumer}: \emph{offset} di-\emph{commit} setelah pemrosesan, sehingga pesan yang belum selesai diproses akan dikirim ulang setelah \emph{consumer} hidup kembali.
 \item \textbf{Konsekuensinya, \emph{consumer} harus idempotent.} Kelompok menanganinya dengan membuang duplikat berdasarkan \code{event_id}, yang bersifat deterministik (\code{hazard_id:version}).
\end{itemize}

Pola yang dipakai adalah \emph{publish-subscribe} dengan satu \emph{consumer group} untuk tiap pihak yang berlangganan. Jika dua
\emph{consumer} dimasukkan ke group yang sama, Kafka akan \emph{membagi} \emph{event} di antara mereka (pola antrean kerja), bukan
menyalinnya. Karena tiap pihak harus menerima \emph{semua} \emph{event}, kami memberi tiap pihak group sendiri.

Untuk semantik pengiriman, \emph{at-most-once} ditolak karena \emph{event} peringatan bisa hilang tanpa diketahui, dan itu tidak
dapat diterima untuk data kebencanaan. \emph{Exactly-once} ditolak karena menuntut transaksi dari ujung ke ujung dan jauh
lebih rumit. \emph{Consumer} yang idempotent menghasilkan efek yang sama dengan biaya yang jauh lebih kecil.

Konsekuensi memilih Kafka adalah bebannya yang lebih berat dibanding RabbitMQ atau NATS (membutuhkan memori JVM, dengan
heap dibatasi 256--512~MB di Compose) dan operasinya yang lebih rumit. Untuk kebutuhan menyusul setelah mati dan
menambah \emph{consumer} tanpa mengubah \emph{producer}, harga itu kami nilai layak.

\section{Batasan yang disadari}

\begin{itemize}
 \item Pembuangan duplikat hanya dilakukan di memori. Kumpulan \code{event_id} hilang saat \emph{consumer} di-\emph{restart}, dan entri lama tidak pernah dihapus (memori tumbuh seiring jumlah \emph{event}). Setelah \emph{restart}, pesan yang belum di-\emph{commit} bisa diproses lagi. Pada M1 \emph{consumer} hanya mencatat dan menghitung sehingga aman. Untuk aksi nyata (misalnya notifikasi sungguhan), pemrosesan ulang harus aman diulang atau memakai penyimpanan dedup yang tahan lama.
 \item Retensi 7~hari. \emph{Consumer} yang mati lebih lama dari itu kehilangan \emph{event} yang sudah kedaluwarsa.
 \item Group baru membaca semua riwayat yang masih tersimpan. Hal ini kami periksa secara terpisah dari skrip demo: group baru memproses 153 \emph{event}, sama dengan jumlah di dashboard. Perilaku ini berguna untuk \emph{consumer} yang perlu mengejar data, tetapi \emph{consumer} yang hanya menginginkan \emph{event} baru harus mengaturnya sendiri.
 \item Hanya ada satu \emph{broker} dengan replikasi 1. Jika disk \emph{broker} hilang, \emph{event} yang belum dibaca ikut hilang. Replikasi dan penanganan kegagalan bersamaan menjadi bahan untuk M2.
 \item Urutan hanya terjamin per hazard (satu key, satu partisi), bukan urutan global lintas hazard.
 \item Pesan yang rusak dilewati dan dicatat di \emph{log} (\code{consumer_bad_message}). Belum ada \emph{dead-letter queue}.
 \item \emph{Consumer} pada M1 bersifat tiruan: dashboard dan notifier hanya mencatat dan menghitung \emph{event}, belum menggerakkan tampilan atau mengirim notifikasi sungguhan.
 \item Hanya ada satu \emph{relay} untuk satu Aggregator. Jika Aggregator dijalankan dalam beberapa \emph{instance} (M2), pembacaan \emph{outbox} perlu dikunci agar \emph{event} tidak terkirim ganda.
 \item \emph{Consumer} untuk portal pemerintah daerah (\code{pemda-portal}) hanya ada saat demo dan tidak termasuk dalam Compose sehari-hari.
\end{itemize}

\section{Jawaban}

\begin{jawaban}[\emph{Broker} dan pola yang dipilih]
\emph{Broker} yang dipilih adalah \textbf{Apache Kafka} (mode KRaft, satu \emph{broker}) dengan pola \emph{publish-subscribe}: satu
\emph{topic} \code{hazard.events} dan satu \emph{consumer group} untuk tiap pihak yang berlangganan. Kafka dipilih karena pesannya
disimpan, sehingga \emph{consumer} yang mati dapat menyusul dan group baru dapat ditambahkan tanpa mengubah \emph{producer}. \emph{Producer}
menulis lewat \emph{transactional outbox} agar tidak ada \emph{event} yang hilang saat \emph{broker} sedang mati.
\end{jawaban}

\begin{jawaban}[Semantik pengiriman dan konsekuensi bagi \emph{consumer}]
Semantik yang berlaku adalah \textbf{at-least-once}. \emph{Producer} menunggu pengakuan penuh (\code{acks=all}, idempotent),
\emph{relay} menandai terkirim setelah pengakuan, dan \emph{consumer} meng-\emph{commit} \emph{offset} setelah memproses. \emph{Event} tidak hilang, tetapi
dapat tiba lebih dari sekali. Karena itu \textbf{\emph{consumer} harus idempotent}. Kelompok menanganinya dengan membuang
duplikat berdasarkan \cmd{event_id = hazard_id:version}, yang bersifat deterministik. Batasannya, kumpulan dedup berada
di memori dan hilang saat \emph{restart}, sehingga aksi sungguhan pada \emph{consumer} di kemudian hari harus aman diulang.
\end{jawaban}
