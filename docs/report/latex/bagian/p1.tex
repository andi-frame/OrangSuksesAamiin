\chapter{Problem 1 -- Arsitektur dan Interoperabilitas Data}

\begin{ringkas}
PVMBG bisa menambah field baru kapan saja tanpa memberi tahu siapa pun. Gateway kami \textbf{tidak menganggap
skema sumber tetap}. Field yang dikenal diperiksa ketat, field lain \textbf{diteruskan apa adanya ke \code{attributes}},
dan hanya perubahan yang mengubah \emph{arti} kontrak (field wajib hilang, tipe berubah, nilai enum baru) yang
\textbf{dikarantina}, bukan ditebak. Semua ini dibuktikan saat sistem berjalan, tanpa restart.
\end{ringkas}

\section{Rancangan solusi dan mekanisme}

\textbf{Mekanisme: \emph{tolerant reader} dengan kebijakan \emph{additive-only}.} Aggregator tidak pernah membaca data
sumber ke struktur (\emph{struct}) yang tetap. Setiap record dibaca ke peta generik (\code{map[string]any}), lalu:

\begin{enumerate}
 \item \textbf{Field yang punya padanan kanonik} dibaca secara eksplisit dan divalidasi (harus ada dan bertipe benar), misalnya \code{magnitude}, \code{epicenter_lat}, \code{occurred_at}, \code{alert_level}, \code{volcano_id}.
 \item \textbf{Setiap field lain} diteruskan apa adanya ke \code{attributes}. Ini mencakup field kontrak yang memang hanya menjadi atribut (\code{depth_km}, \code{eruption_count_24h}) dan field yang muncul belakangan (\code{confidence_level}, atau apa pun).
 \item \textbf{Record yang melanggar kontrak dengan cara yang tidak didukung} tidak ditebak. Record itu dikarantina: dicatat di log (\code{mapping_rejected}), dihitung, dan dapat dilihat di \cmd{GET /internal/quarantine}. Sisa batch tetap diproses dan penanda kemajuan polling (\emph{watermark}) tetap maju.
\end{enumerate}

\textbf{Kebijakan field tak dikenal (dideklarasikan): diteruskan ke \code{attributes}.} Kebijakan ini diterapkan di satu
tempat saja (fungsi \code{splitExtras}), sehingga konsisten untuk semua tipe JSON dan kedua sumber. Demo memeriksanya
dengan pembanding independen: isi \code{attributes} harus \emph{persis sama} dengan himpunan field sumber yang tidak
punya padanan kanonik, tidak ada yang hilang dan tidak ada yang dikarang.

\textbf{Pembaruan record yang sama digabung secara aditif.} Bila record yang sama datang lagi, nilai yang dikirim sumber
sekarang menang, tetapi field yang berhenti dikirim tetap tersimpan. Tanpa aturan ini, mengembalikan skema sumber ke
semula akan menimpa record terbaru dengan versi yang lebih miskin dan membuang field yang sudah sempat ditangkap. Bug
ini sempat muncul saat pengujian dan kini dijaga oleh \code{TestSchemaRollbackDoesNotEraseCapturedFields}.

\textbf{Severity seismik.} AWAS bila ada warning dengan \cmd{threat_level = Awas}. Selain itu mengikuti
\code{threat_level} warning terkait. Tanpa warning: NORMAL untuk magnitudo di bawah 5,0; WASPADA untuk 5,0 sampai
kurang dari 6,5; SIAGA untuk 6,5 ke atas. Bila ada beberapa warning, yang tertinggi menang.

\textbf{Warning tsunami datang belakangan.} Warning diambil dari endpoint terpisah dan sering terbit setelah gempanya
tersimpan. Aggregator mengaitkannya lewat \code{related_event_id}, menghitung ulang severity, dan menyimpannya sebagai
\emph{versi baru} (\code{version} naik dan event \code{updated} diterbitkan). Warning yang gempanya belum terlihat
menunggu (dibatasi TTL) dan menahan watermark warning supaya diambil ulang, bukan hilang. Karena \code{hazard_id}
deterministik dan penyimpanan membandingkan isi, polling ulang pada jendela \emph{overlap} tidak menaikkan versi.

\section{Alasan keputusan dan trade-off}\label{sec:p1-alasan}

Ada beberapa cara untuk menghadapi skema sumber yang berubah. Tabel berikut membandingkannya dengan pilihan kami.

\begin{table}[!htbp]
\centering\small
\begin{tabularx}{\textwidth}{@{}L{3.6cm}Y@{}}
\toprule
\textbf{Alternatif} & \textbf{Pertimbangan}\\
\midrule
Struct tetap & Sederhana dan bertipe, tetapi field baru hilang diam-diam, atau pesan gagal dibaca seluruhnya. Persis masalah di soal.\\
Schema registry & Validasi terpusat. Tetapi menuntut sumber ikut mendaftarkan skema, sedangkan PVMBG mengubah skema \emph{tanpa memberi tahu siapa pun}. Registry tidak akan pernah mutakhir tepat waktu.\\
Protobuf atau Avro (schema evolution) & Evolusi terformalisasi, tetapi sumber mengirim JSON biasa. Lapisan konversi tetap harus tahu skema baru.\\
Pemetaan dinamis lewat berkas konfigurasi & Tanpa redeploy kode, tetapi seseorang tetap harus mengubah konfigurasi saat field baru muncul. Itu masih intervensi manual.\\
\textbf{Tolerant reader + additive-only (dipilih)} & Bekerja \emph{sepihak} dari sisi BNPB: tanpa koordinasi, tanpa redeploy, perilakunya deterministik dan mudah dibuktikan.\\
\bottomrule
\end{tabularx}
\caption{Alternatif mekanisme untuk evolusi skema.}
\end{table}

\textbf{Mengapa ini.} Sumber bersifat otonom dan tidak berkoordinasi, jadi mekanisme harus bekerja tanpa kerja sama
mereka. Perubahan aditif adalah jenis perubahan yang paling umum dan paling aman ditangani otomatis. Perubahan yang
mengubah arti kontrak (ganti nama, ganti tipe, enum baru) sengaja dihentikan di pintu: menebaknya berisiko salah
menafsirkan data kebencanaan. Di sisi penyimpanan, \code{attributes} adalah dokumen JSONB, sehingga kebijakan ini tidak
berubah menjadi migrasi tabel (Problem~4).

\section{Batasan yang disadari}

\begin{itemize}
 \item \textbf{Tipe nilai di \code{attributes} tidak divalidasi.} Ini disengaja (tanpa skema). Perubahan tipe pada field atribut lolos apa adanya.
 \item \textbf{Field tak dikenal ikut diteruskan}, termasuk yang mungkin sensitif. Ini aman karena seluruh \code{attributes} berklasifikasi Mentah dan Media tidak pernah menerimanya (ditegakkan Client-Facing API, Problem~3).
 \item \textbf{Field yang dihapus sumber tetap menempel pada record lama} akibat penggabungan aditif. Hanya record itu; record baru tidak memilikinya. Ini harga agar informasi tidak pernah dibuang.
 \item \textbf{Ganti nama tidak dikenali sebagai ganti nama:} field lama hilang (karantina bila wajib) dan field baru muncul sebagai atribut.
 \item \textbf{Perubahan arti dan satuan tidak terdeteksi.} Tidak ada pemantauan drift selain log dan penghitung karantina.
 \item \textbf{Karantina hanya di memori} (200 entri terakhir, penghitung sejak start). Ia alat diagnosis, bukan penyimpanan resmi, dan belum ada fitur memutar ulang record yang dikarantina setelah kode diperbaiki.
 \item Record yang ditolak diingat 15 menit agar polling ulang pada jendela \emph{overlap} tidak mencatatnya berulang.
 \item Pencocokan enum tidak peka huruf besar-kecil (\code{SIAGA} sama dengan \code{Siaga}); nilai lain ditolak.
\end{itemize}

\section{Jawaban}

\begin{jawaban}
\textbf{Mekanisme yang dipilih:} \emph{tolerant reader} dengan kebijakan \emph{additive-only}. JSON dibaca generik, field
kanonik divalidasi eksplisit, sisanya diteruskan ke \code{attributes}. Schema registry dan Protobuf/Avro tidak dipilih
karena sumber tidak berkoordinasi (alasan di bagian~\ref{sec:p1-alasan}).

\textbf{Kebijakan field tak dikenal:} diteruskan ke \code{attributes} (tidak diabaikan).

\textbf{Didukung:} penambahan field bertipe apa pun; hilangnya field yang hanya berstatus atribut; \code{volcano_id}
baru (turun mutu terkendali).

\textbf{Sengaja tidak didukung} (masuk karantina, batch tetap berlanjut): penghapusan atau penggantian nama field wajib,
perubahan tipe field wajib, dan nilai enum baru. Perubahan arti atau satuan tidak terdeteksi.
\end{jawaban}
